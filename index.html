PRACTICAL-2

Aim: Implement a basic TCP client-server model using Python on Kali Linux.

PULSES:

1) Login with a valid user account. 2) Open Terminal (Ctrl+Alt+T). 3) Verify Python version using python3 --version. 4) Create server.py and client.py files. 5) Run server first, then execute client.

Program:

# --- server.py ---
import socket
server = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
server.bind(('127.0.0.1', 5000))
server.listen(1)
print("Server is listening ...")
conn, addr = server.accept()
print("connected by", addr)
data = conn.recv(1024).decode()
print("Client message:", data)
conn.send("hello from server".encode())
conn.close()

# --- client.py ---
import socket
client = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
client.connect(('127.0.0.1', 5000))
client.send("hello from client".encode())
data = client.recv(1024).decode()
print("server response:", data)
client.close()

PRACTICAL-4

Aim: Implement IPC using pipes or message queues in Python/C on Kali Linux.

PULSES:

• Inter-Process Communication (IPC) • Pipes (Unidirectional) • Parent and Child process • os.pipe() and os.fork() • Parent writes data into the pipe, Child reads data from the pipe • Display received message.

Program:

import os

# Create a unidirectional pipe
read_end, write_end = os.pipe()

# Create a child process
pid = os.fork()

if pid > 0:
    # --- Parent Process ---
    os.close(read_end) # Close reading end
    msg = "Data from parent process via Pipe."
    print("[Parent] Writing data into pipe...")
    os.write(write_end, msg.encode())
    os.close(write_end)
else:
    # --- Child Process ---
    os.close(write_end) # Close writing end
    print("[Child] Reading data from pipe...")
    received = os.read(read_end, 1024).decode()
    print("[Child Received]:", received)
    os.close(read_end)

Output:

[Parent] Writing data into pipe... [Child] Reading data from pipe... [Child Received]: Data from parent process via Pipe.

PRACTICAL-5

Aim: Demonstrate synchronous and asynchronous communication.

PULSES:

• Inter-Process Communication (IPC)
• Synchronous communication (Blocking) vs Asynchronous communication (Non-blocking)
• Part A: Create pipe using os.pipe(), fork() process, write data, Child blocks and waits.
• Part B: Set read end to non-blocking mode, Child continues even if data not available.

Program:

import os, fcntl, time

# Setup a communication pipe
read_end, write_end = os.pipe()

# --- PART A: Synchronous / Blocking Simulation ---
print("[SYNC] Child: Waiting for data (Blocking Read)...")
# Note: In a real run, os.read() pauses the program until data exists
# We simulate the delay to show the "waiting" state.
time.sleep(1)
os.write(write_end, b"Sync_Message")
data = os.read(read_end, 100).decode()
print(f"[SYNC] Child: Received -> {data}\n")

# --- PART B: Asynchronous / Non-blocking Simulation ---
# Change the file descriptor flags to Non-Blocking
flags = fcntl.fcntl(read_end, fcntl.F_GETFL)
fcntl.fcntl(read_end, fcntl.F_SETFL, flags | os.O_NONBLOCK)

print("[ASYNC] Child: Checking for data immediately (Non-blocking)...")
try:
    data = os.read(read_end, 100) # This will not wait
except BlockingIOError:
    print("[ASYNC] Result: No data available. Child continues other execution tasks.")

Output:

[SYNC] Child: Waiting for data (Blocking Read)... (1 second pause observed) [SYNC] Child: Received -> Sync_Message [ASYNC] Child: Checking for data immediately (Non-blocking)... [ASYNC] Result: No data available. Child continues other execution tasks.

Practical 8: Cristian’s and Berkeley Algorithms

import time
import random

# --- 1. Cristian's Algorithm ---
def cristian_algorithm():
    print("--- Cristian's Algorithm ---")
    t0 = time.time()
    time.sleep(random.uniform(0.1, 0.5)) # Simulated network delay
    server_clock = time.time()
    t1 = time.time()
    
    round_trip_delay = t1 - t0
    adjusted_time = server_clock + (round_trip_delay / 2)
    
    print(f"Server Clock: {server_clock}")
    print(f"Adjusted Client Time: {adjusted_time}\n")

# --- 2. Berkeley Algorithm ---
def berkeley_algorithm():
    print("--- Berkeley Algorithm ---")
    clocks = [time.time() + random.randint(-5, 5) for _ in range(3)]
    print(f"Initial Clock Times: {clocks}")
    
    average_time = sum(clocks) / len(clocks)
    print(f"Average Time: {average_time}")
    
    print("Adjusted Clock Times:")
    for i, clock in enumerate(clocks):
        clocks[i] = average_time
        print(f"Machine {i+1}: {clocks[i]}")

Practical 11: Round Robin Scheduling

def round_robin(processes, burst_time, time_quantum):
    n = len(processes)
    remaining_time = list(burst_time)
    current_time = 0
    
    while True:
        done = True
        for i in range(n):
            if remaining_time[i] > 0:
                done = False
                if remaining_time[i] > time_quantum:
                    current_time += time_quantum
                    remaining_time[i] -= time_quantum
                    print(f"{processes[i]} executed for {time_quantum} units")
                else:
                    current_time += remaining_time[i]
                    print(f"{processes[i]} executed for {remaining_time[i]} units")
                    remaining_time[i] = 0
        if done:
            break
    print(f"Total time: {current_time}")

# Input
proc = ["P1", "P2", "P3"]
burst = [10, 5, 8]
quantum = 3
round_robin(proc, burst, quantum)
Practical 12: Ricart-Agrawala Algorithm

import multiprocessing
import time
import random

def process(pid, total_processes, request_queue):
    logical_clock = 0
    time.sleep(random.randint(1, 3))
    logical_clock += 1
    timestamp = logical_clock
    
    # Send requests
    for i in range(total_processes):
        if i != pid:
            request_queue.put(("Request", timestamp, pid))
    
    replies = 0
    while replies < total_processes - 1:
        msg = request_queue.get()
        if msg[0] == "OK" and msg[2] == pid:
            replies += 1
            
    print(f"Process {pid} entering Critical Section")
    time.sleep(2)
    print(f"Process {pid} exiting Critical Section")
    
    # Send OK to others
    for i in range(total_processes):
        if i != pid:
            request_queue.put(("OK", logical_clock, i))

if __name__ == "__main__":
    total = 3
    q = multiprocessing.Queue()
    procs = []
    for p_id in range(total):
        p = multiprocessing.Process(target=process, args=(p_id, total, q))
        procs.append(p)
        p.start()
        
    for p in procs:
        p.join()

Practical 14: Wait-For Graph (Deadlock Detection)

from collections import defaultdict

class WaitForGraph:
    def __init__(self):
        self.graph = defaultdict(list)

    def add_edge(self, p1, p2):
        self.graph[p1].append(p2)

    def dfs(self, node, visited, rec_stack):
        visited.add(node)
        rec_stack.add(node)
        
        for neighbor in self.graph[node]:
            if neighbor not in visited:
                if self.dfs(neighbor, visited, rec_stack):
                    return True
            elif neighbor in rec_stack:
                return True
        
        rec_stack.remove(node)
        return False

    def detect_deadlock(self):
        visited = set()
        rec_stack = set()
        for process in list(self.graph):
            if process not in visited:
                if self.dfs(process, visited, rec_stack):
                    return True
        return False

# Usage
wfg = WaitForGraph()
n = int(input("Enter number of edges: "))
print("Enter in format: P1 P2")
for _ in range(n):
    p1, p2 = input().split()
    wfg.add_edge(p1, p2)

print("\nWait-For Graph:")
for proc in wfg.graph:
    print(f"{proc} -> {wfg.graph[proc]}")

if wfg.detect_deadlock():
    print("Deadlock Detected")
else:
    print("No Deadlock")
